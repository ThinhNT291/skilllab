// Lấy (hoặc tạo mới) folder Drive DÙNG CHUNG cho mọi loại nội dung tĩnh — giống hệt getTTSCacheFolder()
// ở Controller_Shadowing.gs, lưu folderId vào PropertiesService để không phải tìm/tạo lại mỗi lần.
function getContentStoreFolder_() {
  let props = PropertiesService.getScriptProperties();
  let folderId = props.getProperty('CONTENT_STORE_FOLDER_ID');
  if (folderId) {
    try { return DriveApp.getFolderById(folderId); }
    catch (e) { /* folder bị xoá tay -> rơi xuống tạo lại bên dưới */ }
  }
  let folder = DriveApp.createFolder('TrainingZone_ContentStore_DoNotDelete');
  props.setProperty('CONTENT_STORE_FOLDER_ID', folder.getId());
  return folder;
}

// Tối đa CacheService cho phép (đã tra chính thức, xem ghi chú đầu file) — đặt hằng số riêng để mọi
// contentType dùng chung, không lặp lại con số rải rác nhiều nơi.
const CONTENT_STORE_CACHE_TTL_SEC = 21600; // 6 giờ

function contentStoreFileName_(contentType, key) {
  return contentType + '__' + key + '.json';
}
function contentStoreCacheKey_(contentType, key) {
  return 'cs_' + contentType + '_' + key;
}

// Đọc 1 nội dung: ưu tiên CacheService (nhanh, không tốn lượt gọi Drive); cache miss/hết hạn (>6 giờ)
// mới đọc Drive rồi tự nạp lại cache cho lượt sau. Trả về null nếu key này CHƯA TỪNG được ghi qua
// putContent_() (vd chưa chạy migration, hoặc key gõ sai) — bên gọi tự quyết định xử lý sao (báo lỗi/
// rơi về đường dự phòng), hàm này không throw để không làm gãy luồng chính.
function getCachedContent_(contentType, key) {
  let cache = CacheService.getScriptCache();
  let cacheKey = contentStoreCacheKey_(contentType, key);
  let cached = cache.get(cacheKey);
  if (cached) {
    try { return JSON.parse(cached); }
    catch (e) { /* cache hỏng (hiếm) -> rơi xuống đọc lại Drive như cache miss bình thường */ }
  }

  let folder = getContentStoreFolder_();
  let fileName = contentStoreFileName_(contentType, key);
  let files = folder.getFilesByName(fileName);
  if (!files.hasNext()) return null;

  let jsonText = files.next().getBlob().getDataAsString();
  try { cache.put(cacheKey, jsonText, CONTENT_STORE_CACHE_TTL_SEC); }
  catch (e) { Logger.log('Không nạp được cache (bỏ qua, vẫn trả đúng dữ liệu từ Drive): ' + e); }
  return JSON.parse(jsonText);
}

// Ghi/ghi đè 1 nội dung — dùng khi soạn/sửa nội dung (migration script hoặc công cụ soạn nội dung sau
// này). AN TOÀN GỌI LẠI NHIỀU LẦN (idempotent): luôn xoá hết file CŨ trùng tên trước khi tạo file mới,
// tránh Drive tích tụ nhiều bản trùng tên gây đọc nhầm bản cũ; đồng thời xoá cache liên quan để lần đọc
// kế tiếp lấy ngay bản mới, không phải đợi cache tự hết hạn.
function putContent_(contentType, key, dataObject) {
  let folder = getContentStoreFolder_();
  let fileName = contentStoreFileName_(contentType, key);
  let existingFiles = folder.getFilesByName(fileName);
  while (existingFiles.hasNext()) existingFiles.next().setTrashed(true);

  let jsonText = JSON.stringify(dataObject);
  folder.createFile(fileName, jsonText, MimeType.PLAIN_TEXT);

  CacheService.getScriptCache().remove(contentStoreCacheKey_(contentType, key));
}

// Xoá tường minh 1 mục khỏi cache (không đụng tới Drive) — dùng khi biết chắc nội dung đã đổi mà không
// qua putContent_() (vd cần buộc đọc lại ngay để kiểm tra, xem verifyGrammarContentStore_() ở
// Migration_GrammarToDrive.gs).
function invalidateContentCache_(contentType, key) {
  CacheService.getScriptCache().remove(contentStoreCacheKey_(contentType, key));
}
